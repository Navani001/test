Enum indicator_category {
  CLINICAL
  ADMINISTRATIVE
  DIAGNOSTIC
  PROCEDURAL
  OTHER
}

Enum indicator_output {
  BOOLEAN
  TEXT
  NUMBER
  DATE
  ENUM
  JSON
}


// ============================================================
// USER
// ============================================================

Table users {
  id uuid [pk]

  name varchar
  username varchar [unique]
  password varchar

  is_active boolean

  created_at timestamp
  updated_at timestamp
}


// ============================================================
// CLIENT
// ============================================================

Table clients {
  id uuid [pk]

  name varchar
  schema_name varchar
  crd_id int

  is_base boolean
  is_active boolean

  created_at timestamp
  updated_at timestamp
}


// ============================================================
// CONDITION
// ============================================================

Table conditions {
  id uuid [pk]

  current_version_id uuid [null]

  key varchar

  is_active boolean
  user_id uuid

  created_at timestamp
  updated_at timestamp
}


// ============================================================
// CONDITION - CLIENT
// ============================================================

Table condition_clients {
  condition_id uuid [not null]
  client_id uuid [not null]

  indexes {
    (condition_id, client_id) [pk]
  }
}


// ============================================================
// CONDITION VERSION
// ============================================================

Table condition_versions {
  id uuid [pk]

  condition_id uuid [not null]

  version int [not null]
  display_name varchar

  is_draft boolean
  is_active boolean

  created_by uuid [not null]

  created_at timestamp
  updated_at timestamp

  indexes {
    (condition_id, version) [unique]
  }
}


// ============================================================
// INDICATOR
// ============================================================

Table indicators {
  id uuid [pk]

  current_version_id uuid [null]

  key varchar

  is_active boolean

  created_at timestamp
  updated_at timestamp
}


// ============================================================
// INDICATOR VERSION
// ============================================================

Table indicators_versions {
  id uuid [pk]

  indicator_id uuid [not null]

  name varchar

  category indicator_category
  output indicator_output

  cui_code varchar [null]

  version int [null]

  is_active boolean
  is_draft boolean

  created_at timestamp
  updated_at timestamp
}


// ============================================================
// CONDITION - INDICATOR
// ============================================================

Table condition_indicators {
  condition_version_id uuid [not null]
  indicator_id uuid [not null]

  indexes {
    (condition_version_id, indicator_id) [pk]
  }
}


// ============================================================
// SECTION
// ============================================================

Table sections {
  id uuid [pk]

  name varchar

  created_at timestamp
  updated_at timestamp
}


// ============================================================
// INDICATOR - SECTION
// ============================================================

Table indicator_sections {
  indicator_version_id uuid [not null]
  section_id uuid [not null]

  indexes {
    (indicator_version_id, section_id) [pk]
  }
}


// ============================================================
// INDICATOR ELEMENT
// ============================================================

Table indicator_elements {
  id uuid [pk]

  indicator_version_id uuid [not null]

  name varchar

  created_at timestamp
  updated_at timestamp
}


// ============================================================
// ICD CODE
// ============================================================

Table icd_codes {
  id uuid [pk]

  code varchar [unique]
  description varchar

  is_active boolean

  created_at timestamp
  updated_at timestamp
}


// ============================================================
// CONDITION - ICD CODE
// ============================================================

Table condition_icd_codes {
  condition_version_id uuid [not null]
  icd_code_id uuid [not null]

  indexes {
    (condition_version_id, icd_code_id) [pk]
  }
}


// ============================================================
// ATOM
// ============================================================

Table atoms {
  id uuid [pk]

  name varchar
  description text

  is_active boolean

  created_at timestamp
  updated_at timestamp
}


// ============================================================
// INDICATOR VERSION - ATOM
// ============================================================

Table indicator_atoms {
  indicator_version_id uuid [not null]
  atom_id uuid [not null]

  indexes {
    (indicator_version_id, atom_id) [pk]
  }
}


// ============================================================
// AUDIT LOG
// ============================================================

Table audit_logs {
  id uuid [pk]

  user_id uuid [null]

  entity_type varchar [not null]
  entity_key varchar [not null]

  action varchar [not null]

  old_data json
  new_data json

  created_at timestamp
}
