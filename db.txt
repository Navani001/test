from datetime import datetime, timezone

from sqlalchemy import select

from s3_service import S3Service

# Your existing SQLAlchemy session
from database import SessionLocal

# Your existing model
from models import LogFileStatus

# Your existing parser
from parser import parse_log

# Your existing DB update function
from database_operations import update_database


S3_BUCKET = "your-bucket-name"

S3_PREFIX = "logs/"

AWS_REGION = "ap-south-1"


def get_log_status(
    session,
    s3_key: str,
    etag: str,
):

    return session.scalar(
        select(LogFileStatus).where(
            LogFileStatus.s3_key == s3_key,
            LogFileStatus.etag == etag,
        )
    )


def create_or_get_log_status(
    session,
    s3_key: str,
    etag: str,
):

    log_status = get_log_status(
        session,
        s3_key,
        etag,
    )

    if log_status:
        return log_status

    log_status = LogFileStatus(
        s3_key=s3_key,
        etag=etag,
        status="PENDING",
        attempts=0,
    )

    session.add(log_status)

    session.flush()

    return log_status


def process_logs():

    s3 = S3Service(
        bucket_name=S3_BUCKET,
        prefix=S3_PREFIX,
        region_name=AWS_REGION,
    )

    with SessionLocal() as session:

        for log in s3.list_logs():

            s3_key = log["key"]
            etag = log["etag"]

            print(
                f"Checking S3 log: {s3_key}"
            )

            # -----------------------------------------
            # Get status from database
            # -----------------------------------------

            log_status = get_log_status(
                session,
                s3_key,
                etag,
            )

            # -----------------------------------------
            # Already processed
            # -----------------------------------------

            if (
                log_status
                and log_status.status == "COMPLETED"
            ):

                print(
                    f"Already processed: {s3_key}"
                )

                continue

            try:

                # -------------------------------------
                # Create tracking record if necessary
                # -------------------------------------

                if log_status is None:

                    log_status = create_or_get_log_status(
                        session,
                        s3_key,
                        etag,
                    )

                # -------------------------------------
                # Mark PROCESSING
                # -------------------------------------

                log_status.status = "PROCESSING"

                log_status.attempts += 1

                log_status.error_message = None

                session.commit()

                # -------------------------------------
                # Download from S3
                # -------------------------------------

                print(
                    f"Downloading: {s3_key}"
                )

                content = s3.download_log(
                    s3_key
                )

                # -------------------------------------
                # YOUR EXISTING PARSER
                # -------------------------------------

                parsed_data = parse_log(
                    content
                )

                # -------------------------------------
                # YOUR EXISTING DATABASE UPDATE
                # -------------------------------------

                update_database(
                    session,
                    parsed_data,
                )

                # -------------------------------------
                # Mark COMPLETED
                # -------------------------------------

                log_status.status = "COMPLETED"

                log_status.processed_at = (
                    datetime.now(timezone.utc)
                )

                log_status.error_message = None

                session.commit()

                print(
                    f"Successfully processed: "
                    f"{s3_key}"
                )

            except Exception as exc:

                # -------------------------------------
                # Rollback parser/database transaction
                # -------------------------------------

                session.rollback()

                print(
                    f"Failed processing "
                    f"{s3_key}: {exc}"
                )

                # -------------------------------------
                # Update status to FAILED
                # -------------------------------------

                log_status = get_log_status(
                    session,
                    s3_key,
                    etag,
                )

                if log_status:

                    log_status.status = "FAILED"

                    log_status.error_message = str(
                        exc
                    )[:5000]

                    session.commit()


if __name__ == "__main__":
    process_logs()
